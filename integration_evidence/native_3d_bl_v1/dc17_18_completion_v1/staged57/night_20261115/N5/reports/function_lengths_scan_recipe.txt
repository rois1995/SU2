def mask_cpp(s):
    o=list(s);i=0;state='code';quote=''
    while i<len(s):
        if state=='code':
            if s.startswith('//',i):o[i]=o[i+1]=' ';i+=2;state='line';continue
            if s.startswith('/*',i):o[i]=o[i+1]=' ';i+=2;state='block';continue
            if s[i] in ('"',"'"):quote=s[i];o[i]=' ';i+=1;state='str';continue
            i+=1
        elif state=='line':
            if s[i]=='\n':state='code'
            else:o[i]=' '
            i+=1
        elif state=='block':
            if s.startswith('*/',i):o[i]=o[i+1]=' ';i+=2;state='code'
            else:
                if s[i]!='\n':o[i]=' '
                i+=1
        else:
            if s[i]=='\\':
                o[i]=' '
                if i+1<len(s) and s[i+1]!='\n':o[i+1]=' '
                i+=2
            elif s[i]==quote:o[i]=' ';i+=1;state='code'
            else:
                if s[i]!='\n':o[i]=' '
                i+=1
    return ''.join(o)

def balanced_end(mask, open_at):
    depth=0
    for i in range(open_at,len(mask)):
        if mask[i]=='{':depth+=1
        elif mask[i]=='}':
            depth-=1
            if depth==0:return i
    raise ValueError('unclosed function body')

def cpp_functions(rel):
    path=repo/rel;s=path.read_text();m=mask_cpp(s);out=[]

        names=r'(?:Phase|Seconds|PrintQueries|PrintMoves|Root|~Root|Star|~Star|Candidate|~Candidate|WorkScope|~WorkScope|Measure|Quality|~Quality)'
        pat=re.compile(r'^[ \t]*(?:(?:inline|explicit)\s+)?(?:[\w:<>,*& \t]+?\s+)?('+names+r')\s*\(',re.M)
    else:
        pat=re.compile(r'^[ \t]*(?:inline\s+)?[^\n;{}]*?\b(D9[A-Za-z0-9_]+)\s*\(',re.M)
    for mt in pat.finditer(m):
        name=mt.group(1); start=mt.start(); paren=m.find('(',mt.start(1)+len(name)-1)
        depth=0; close=None
        for j in range(paren,len(m)):
            if m[j]=='(':depth+=1
            elif m[j]==')':
                depth-=1
                if depth==0:close=j;break
        if close is None:continue
        j=close+1; brace=None
        while j<len(m) and m[j] not in '{;':j+=1
        if j>=len(m) or m[j]!='{':continue
        brace=j
        try:end=balanced_end(m,brace)
        except ValueError as e: raise ValueError(f"{rel}:{m.count(chr(10),0,start)+1} {name} brace {brace}") from e
        prefix=m[m.rfind('\n',0,start)+1:mt.start(1)]
        if '=' in prefix or 'return' in prefix or 'CHECK' in prefix or '.' in prefix:continue
        # D9 helper signatures require a return type; RAII/header methods are explicitly enumerated.
        startline=m.count('\n',0,start)+1;endline=m.count('\n',0,end)+1
        out.append({'path':rel,'function':name,'start_line':startline,'end_line':endline,'lines':endline-startline+1,'language':'C++'})
    # unique on name/start; pattern can match nested signature text more than once
    uniq={(r['start_line'],r['function']):r for r in out}
    return list(uniq.values())
