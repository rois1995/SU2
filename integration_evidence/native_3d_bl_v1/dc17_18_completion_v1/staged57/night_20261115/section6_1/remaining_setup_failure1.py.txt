"""Load the accepted safeguard runner with section 6.1 paths and build522."""
import hashlib
import json
import ast
import types
import os
import sys
from pathlib import Path

HERE = Path(__file__).resolve().parent
STAGED = HERE.parents[1]
REPO = STAGED.parents[3]
OLD = STAGED / "p7p8_r3_20261115/run_remaining_checks.py"
GEOMETRY = STAGED / "check_geometry_magnitude481.py"
SAVED = STAGED / "check_saved_surface_corrections481.py"


def sha(path):
    """Hash one pinned validation input."""
    with Path(path).open("rb") as stream:
        return hashlib.file_digest(stream, "sha256").hexdigest()


def replace_once(source, before, after):
    """Require one reviewed source anchor before retargeting it."""
    assert source.count(before) == 1, f"expected one source anchor: {before!r}"
    return source.replace(before, after, 1)


def adapted_source(mode=None):
    """Retarget the accepted safeguard runner while preserving its assertions."""
    source = OLD.read_text()
    source = replace_once(
        source,
        "HERE=Path(__file__).resolve().parent;R=HERE.parent;REPO=R.parents[3]",
        "HERE=Path(__file__).resolve().parent;STAGED=HERE.parents[1];R=STAGED;REPO=STAGED.parents[3]",
    )
    source = replace_once(
        source,
        "sys.path.insert(0,str(R));import run_cases as checks",
        "sys.path.insert(0,str(R));import run_cases as checks\n"
        "from build522_aliases import main as expose_build\n"
        "expose_build();checks.ROOT=HERE;checks.REPO=REPO",
    )
    source = source.replace("build508", "build522")
    source = replace_once(source, "(HERE/'build_status.json')",
                          "(HERE/'build522_status.json')")
    source = replace_once(source, "(HERE/'source_pins.json')",
                          "(HERE/'build522_source_pins.json')")
    source = source.replace("R/'build522_status.json'", "HERE/'build522_status.json'")
    source = source.replace("R/'build522_source_pins.json'", "HERE/'build522_source_pins.json'")
    source = source.replace("R/'build522_terminal_handles.json'", "HERE/'build522_terminal_handles.json'")
    source = source.replace("HERE.parent/'purpose_query_fix_20261114", "R/'purpose_query_fix_20261114")
    source = replace_once(source, "gate_path=HERE/'check_cheap_bl.py'",
                          "gate_path=OLD.parent/'check_cheap_bl.py'")
    source = replace_once(source, "extra={}",
                          "extra={str(OLD.resolve()):sha(OLD),str(Path(__file__).resolve()):sha(__file__)}")
    source = replace_once(source, "extra_pins={str(Path(__file__).resolve()):sha(__file__)}",
                          "extra_pins={str(OLD.resolve()):sha(OLD),str(Path(__file__).resolve()):sha(__file__),str((HERE/'build522_aliases.py').resolve()):sha(HERE/'build522_aliases.py')}")
    start = "for path,value in ((HERE/'build522_status.json',metadata)"
    end = "original_save=checks.save"
    assert source.count(start) == 1
    left = source.index(start)
    right = source.index(end, left)
    source = source[:left] + source[right:]
    source = source.replace("case='p7p8_'+mode", "case='section6_1_'+mode")
    if mode == "bl":
        loader = "spec=importlib.util.spec_from_file_location('bl_regression',gate_path)\n   gate=importlib.util.module_from_spec(spec);spec.loader.exec_module(gate)"
        replacement = (
            "gate_source=gate_path.read_text()\n"
            "   assert gate_source.count('20_000_000_000')==1\n"
            "   gate_source=gate_source.replace('20_000_000_000','19_999_490_048',1)\n"
            "   gate_source=gate_source.replace('whole_job_peak_RSS_20GB_serial','whole_job_peak_RSS_19073_MiB_serial')\n"
            "   gate_source=gate_source.replace('20GB serial peak gate','19073 MiB serial peak gate')\n"
            "   gate_namespace={'__name__':'section6_1_bl_gate','__file__':str(gate_path)}\n"
            "   exec(compile(gate_source,str(gate_path),'exec'),gate_namespace)\n"
            "   gate=__import__('types').SimpleNamespace(**gate_namespace)"
        )
        source = replace_once(source, loader, replacement)
    return source


def execute_helper(path, source, extra=None):
    """Execute one existing check with its original inputs and section 6.1 output root."""
    namespace = {"__name__": "__main__", "__file__": str(path),
                 "CASE_ROOT": HERE / "cases", "HERE": HERE}
    if extra:
        namespace.update(extra)
    exec(compile(source, str(path), "exec"), namespace)


def reset_native_environment():
    """Match the house runner's single-threaded, isolated native environment."""
    for key in list(os.environ):
        if key.startswith("SU2_NATIVE_"):
            del os.environ[key]
    os.environ.update(OMP_NUM_THREADS="1", OPENBLAS_NUM_THREADS="1",
                      MKL_NUM_THREADS="1", OMPI_MCA_osc="pt2pt")


def geometry_source():
    """Keep the geometry check inputs and redirect its attempt census."""
    source = GEOMETRY.read_text()
    source = replace_once(source, "prior = set((R / 'cases').glob(CASE + '_*'))",
                          "prior = set(CASE_ROOT.glob(CASE + '_*'))")
    source = replace_once(source, "set((R / 'cases').glob(CASE + '_*')) - prior",
                          "set(CASE_ROOT.glob(CASE + '_*')) - prior")
    return source


def configure_geometry_namespace(namespace):
    """Override globals in the exact dictionary used by the loaded helper."""
    namespace.update(BUILD="build522", CASE="geometry_merit505_serial",
                     STATUS=HERE / "build522_status.json",
                     HANDLES=HERE / "build522_terminal_handles.json",
                     SOURCE_PINS=HERE / "build522_source_pins.json",
                     RESULT=HERE / "geometry_merit505_serial_assessment.json")


def saved_source(mode):
    """Keep the saved-state check inputs and redirect its build and outputs."""
    source = SAVED.read_text()
    source = source.replace("build481", "build522")
    source = source.replace("curved481_saved_surface_corrections_",
                            "curved505_saved_surface_corrections_")
    source = replace_once(source, "assert json.loads((R/'build522_terminal_handles.json').read_text())",
                          "assert json.loads((HERE/'build522_terminal_handles.json').read_text())")
    source = replace_once(source, "attempt=sorted((R/'cases').glob(case+'_*')",
                          "attempt=sorted(CASE_ROOT.glob(case+'_*')")
    source = source.replace("sys.argv=['run_cases.py','build522',case]",
                            "sys.argv=['run_cases.py','build522',case]")
    if mode == "saved_current":
        assert source.count("curved505_saved_surface_corrections_") >= 1
        source = source.replace("curved505_saved_surface_corrections_",
                                "curved505_current_target_surface_corrections_")
    return source


def run_geometry():
    """Run the original geometry cohort against the current pinned binary."""
    reset_native_environment()
    import run_cases as checks
    checks.ROOT = HERE
    checks.REPO = REPO
    status = json.loads((HERE / "build522_status.json").read_text())
    checks.BINARY = Path(status["test_binary"])
    source = geometry_source()
    namespace = {"__name__": "section6_1_geometry_check", "__file__": str(GEOMETRY),
                 "CASE_ROOT": HERE / "cases", "REPO": REPO}
    exec(compile(source, str(GEOMETRY), "exec"), namespace)
    configure_geometry_namespace(namespace)
    original_save = checks.save

    def save_pinned(path, value):
        """Add this wrapper to the house per-case provenance."""
        if Path(path).name == "pins.json":
            value[str(Path(__file__).resolve())] = sha(__file__)
            alias = HERE / "build522_aliases.py"
            value[str(alias.resolve())] = sha(alias)
        original_save(path, value)

    checks.save = save_pinned
    namespace["main"]()


def run_saved(mode):
    """Run one saved-state cohort with its required frozen target."""
    reset_native_environment()
    import run_cases as checks
    checks.ROOT = HERE
    checks.REPO = REPO
    status = json.loads((HERE / "build522_status.json").read_text())
    checks.BINARY = Path(status["test_binary"])
    source = saved_source(mode)
    pinned_inputs = {str(Path(__file__).resolve()): sha(__file__),
                     str(SAVED.resolve()): sha(SAVED),
                     str((HERE / "build522_aliases.py").resolve()):
                         sha(HERE / "build522_aliases.py")}
    if mode == "saved_current":
        producer_receipt = STAGED / "target_checkpoint492_producer.json"
        qualification = STAGED / "target_checkpoint492_qualification.json"
        receipt = json.loads(producer_receipt.read_text())
        qualified = json.loads(qualification.read_text())
        assert qualified["actual_target_checkpoint_qualified"]
        assert qualified["recomposed_target_checked_after_reload"]
        assert qualified["producer_receipt_sha256"] == sha(producer_receipt)
        producer = Path(receipt["producer"])
        assert producer.resolve() == Path(qualified["producer"]).resolve()
        for name, item in receipt["files"].items():
            file = producer / name
            assert sha(file) == item["sha256"]
            pinned_inputs[str(file)] = item["sha256"]
        reference = STAGED / "cases/curved492_saved_production_surface_serial_20261113T163756_3724702"
        exact = json.loads((reference / "command.json").read_text())["environment"]
        current = {key: exact[key] for key in (
            "SU2_NATIVE_TARGET_CHECKPOINT3D", "SU2_NATIVE_FROZEN_SENSOR3D",
            "SU2_NATIVE_FROZEN_CORE3D", "SU2_NATIVE_FROZEN_TANGENT_SCALE3D")}
        pinned_inputs[str(producer_receipt)] = sha(producer_receipt)
        pinned_inputs[str(qualification)] = sha(qualification)
        for variable in ("SU2_NATIVE_TARGET_CHECKPOINT3D", "SU2_NATIVE_FROZEN_SENSOR3D"):
            pinned_inputs[str(Path(current[variable]))] = sha(current[variable])
        original_main = checks.main

        def current_main(runtime_manifest=None):
            """Apply the current target after the legacy fixture sets its default target."""
            os.environ.update(current)
            return original_main(runtime_manifest)

        checks.main = current_main
    original_save = checks.save

    def save_pinned(path, value):
        """Add this wrapper to the house per-case provenance."""
        if Path(path).name == "pins.json":
            value.update(pinned_inputs)
        original_save(path, value)

    checks.save = save_pinned
    sys.argv = [str(SAVED), "serial"]
    execute_helper(SAVED, source, {"OLD": OLD, "STAGED": STAGED,
                                   "REPO": REPO, "checks": checks,
                                   "sha": sha, "HERE": HERE,
                                   "CASE_ROOT": HERE / "cases",
                                   "original_save": original_save})


def current_target_self_check():
    """Exercise the real late callback after a simulated legacy target overwrite."""
    tree = ast.parse(Path(__file__).read_text())
    callback = next(node for node in ast.walk(tree)
                    if isinstance(node, ast.FunctionDef) and node.name == "current_main")
    environment = {"SU2_NATIVE_TARGET_CHECKPOINT3D": "legacy"}
    namespace = {"os": types.SimpleNamespace(environ=environment),
                 "current": {"SU2_NATIVE_TARGET_CHECKPOINT3D": "current492"},
                 "original_main": lambda runtime: dict(environment)}
    module = ast.Module(body=[callback], type_ignores=[])
    exec(compile(module, str(Path(__file__)), "exec"), namespace)
    assert namespace["current_main"]()["SU2_NATIVE_TARGET_CHECKPOINT3D"] == "current492"


def main():
    """Run the selected ordered validation action."""
    source = adapted_source()
    if sys.argv[1:] == ["--self-check"]:
        current_target_self_check()
        geometry = geometry_source()
        saved = saved_source("saved_current")
        bl = adapted_source("bl")
        namespace = {"BUILD": "stale"}
        exec("def read_build():\n    return BUILD", namespace)
        configure_geometry_namespace(namespace)
        assert "checks.ROOT=HERE" in source
        assert "build522_status.json" in source
        assert "OLD.parent/'check_cheap_bl.py'" in source
        assert "R/'purpose_query_fix_20261114" in source
        assert "CASE_ROOT.glob(CASE + '_*')" in geometry
        assert "CASE_ROOT.glob(case+'_*')" in saved
        assert "HERE/'build522_terminal_handles.json'" in saved
        assert "build481" not in saved
        assert "19_999_490_048" in bl
        assert "gate_source.count('20_000_000_000')==1" in bl
        assert namespace["read_build"]() == "build522"
        print(json.dumps({"passed": True, "old_runner_sha256": sha(OLD),
                          "transformed_source_sha256": hashlib.sha256(source.encode()).hexdigest(),
                          "geometry_source_sha256": hashlib.sha256(geometry.encode()).hexdigest(),
                          "saved_current_source_sha256": hashlib.sha256(saved.encode()).hexdigest(),
                          "bl_source_sha256": hashlib.sha256(bl.encode()).hexdigest(),
                          "bl_memory_cap_bytes": 19_999_490_048}))
        return
    mode = sys.argv[1]
    if mode == "merit481":
        run_geometry()
        return
    if mode in ("saved481", "saved_current"):
        run_saved(mode)
        return
    namespace = {"__name__": "__main__", "__file__": str(Path(__file__).resolve()),
                 "OLD": OLD}
    mode = sys.argv[1]
    namespace["SECTION6_MODE"] = mode
    if mode == "bl":
        source = adapted_source("bl")
    exec(compile(source, str(OLD), "exec"), namespace)


if __name__ == "__main__":
    main()
